111 def classify(
112 self,
113 query: str,
114 context: Optional[QueryContext] = None
115 ) -> ComplexityResult:
116 """
117 @brief Классифицирует запрос по уровню сложности.
118
119 Алгоритм:
120 1. Проверяет простые паттерны (приветствия, FAQ) — если нет MEDIUM/COMPLEX слов.
121 2. Считает COMPLEX-ключевые слова; если ≥2 или длина ≥300 → COMPLEX.
122 3. Проверяет MEDIUM-ключевые слова.
123 4. Fallback по длине строки.
124
125 @param query Текст запроса пользователя.
126 @param context Дополнительный контекст (thread_id, user_id) — опционально.
127 @return ComplexityResult с уровнем, уверенностью и рекомендуемой моделью.
128 """
129 query_lower = query.lower().strip()
130 query_length = len(query)
131
132 # Подсчитываем ключевые слова заранее
133 complex_matches = self._count_keyword_matches(query_lower, self.COMPLEX_KEYWORDS)
134 medium_matches = self._count_keyword_matches(query_lower, self.MEDIUM_KEYWORDS)
135 has_medium_keywords = medium_matches >= 1 or self._has_medium_keywords(query_lower)
136
137 # 1. Проверяем простые паттерны (приветствия, FAQ)
138 # Но только если нет MEDIUM/COMPLEX ключевых слов
139 if self._is_simple_pattern(query_lower) and not has_medium_keywords and complex_matches == 0:
140 return self._create_result(
141 ComplexityLevel.SIMPLE,
142 confidence=0.95,
143 reasoning="Простой шаблонный запрос (приветствие/FAQ)"
144 )
145
146 # 2. Проверяем наличие COMPLEX ключевых слов
147 if complex_matches >= 2 or query_length >= self.COMPLEX_MIN_LENGTH:
148 return self._create_result(
149 ComplexityLevel.COMPLEX,
150 confidence=min(0.9, 0.6 + complex_matches * 0.1),
151 reasoning=f"Сложный анализ: {complex_matches} ключевых слов, {query_length} символов"
152 )
153
154 # 3. Проверяем наличие MEDIUM ключевых слов
155 if has_medium_keywords:
156 # Если есть и COMPLEX ключевые слова, повышаем до COMPLEX
157 if complex_matches >= 1:
158 return self._create_result(
159 ComplexityLevel.COMPLEX,
160 confidence=0.7,
161 reasoning="Смешанный запрос с аналитическими элементами"
162 )
163 return self._create_result(
164 ComplexityLevel.MEDIUM,
165 confidence=min(0.85, 0.6 + medium_matches * 0.1),
166 reasoning=f"Средняя сложность: {medium_matches} ключевых слов"
167 )
168
169 # 4. Проверяем очень короткие запросы
170 if query_length <= self.SIMPLE_MAX_LENGTH:
171 return self._create_result(
172 ComplexityLevel.SIMPLE,
173 confidence=0.8,
174 reasoning=f"Короткий запрос ({query_length} символов)"
175 )
176
177 # 5. Классификация по длине (fallback)
178 if query_length <= self.MEDIUM_MAX_LENGTH:
179 return self._create_result(
180 ComplexityLevel.SIMPLE,
181 confidence=0.6,
182 reasoning="Средняя длина без специфичных ключевых слов"
183 )
184 else:
185 return self._create_result(
186 ComplexityLevel.MEDIUM,
187 confidence=0.6,
188 reasoning=f"Длинный запрос ({query_length} символов)"
189 )

191 def _is_simple_pattern(self, query: str) -> bool:
192 """
193 @brief Проверяет, соответствует ли запрос простым паттернам.
194
195 @param query Текст запроса в нижнем регистре.
196 @return True, если найдено совпадение с одним из SIMPLE_PATTERNS.
197 """
198 for pattern in self._simple_patterns:
199 if pattern.search(query):
200 return True
201 return False
202
203 def _has_medium_keywords(self, query: str) -> bool:
204 """
205 @brief Проверяет наличие ключевых слов средней сложности.
206
207 @param query Текст запроса в нижнем регистре.
208 @return True, если хотя бы одно слово из MEDIUM_KEYWORDS найдено.
209 """
210 return any(kw in query for kw in self.MEDIUM_KEYWORDS)
211
212 def _has_complex_keywords(self, query: str) -> bool:
213 """
214 @brief Проверяет наличие ключевых слов высокой сложности.
215
216 @param query Текст запроса в нижнем регистре.
217 @return True, если хотя бы одно слово из COMPLEX_KEYWORDS найдено.
218 """
219 return any(kw in query for kw in self.COMPLEX_KEYWORDS)
220
221 def _count_keyword_matches(self, query: str, keywords: Set[str]) -> int:
222 """
223 @brief Подсчитывает количество найденных ключевых слов в запросе.
224
225 @param query Текст запроса в нижнем регистре.
226 @param keywords Множество ключевых слов для поиска.
227 @return Количество совпадений.
228 """
229 return sum(1 for kw in keywords if kw in query)

231 def _create_result(
232 self,
233 level: ComplexityLevel,
234 confidence: float,
235 reasoning: str
236 ) -> ComplexityResult:
237 """
238 @brief Создаёт объект результата классификации.
239
240 Маппит уровень сложности на имя модели через settings.MODEL_COMPLEXITY_MAP
241 и логирует решение на уровне DEBUG.
242
243 @param level Уровень сложности (SIMPLE / MEDIUM / COMPLEX).
244 @param confidence Уверенность классификатора (0.0–1.0).
245 @param reasoning Текстовое объяснение решения.
246 @return ComplexityResult с заполненными полями.
247 """
248 # Маппинг сложности на модели
249 model_map = settings.MODEL_COMPLEXITY_MAP
250 suggested_model = model_map.get(level.value, "gigachat")
251
252 logger.debug(
253 f"Classified query as {level.value} "
254 f"(confidence={confidence:.2f}) -> {suggested_model}"
255 )
256
257 return ComplexityResult(
258 level=level,
259 confidence=confidence,
260 reasoning=reasoning,
261 suggested_model=suggested_model
262 )
