27 # Regex patterns for PII
28 EMAIL_PATTERN = r'\b[A-Za-z0-9._%+-]+@[A-Za-z0-9.-]+\.[A-Z|a-z]{2,}\b'
29 PHONE_PATTERN = r'\b(?:\+?(\d{1,3}))?[-. (]*(\d{3})[-. )]*(\d{3})[-. ]*(\d{4})\b'
30 # Simple name pattern (capitalized words, simplistic) - use with caution or NLP in production
31 # For now, we'll focus on emails and phones as they are most critical and easier to regex
32
33 def anonymize_text(self, text: str) -> str:
34 """
35 @brief Анонимизирует PII в строке.
36
37 @param text Исходный текст.
38 @return Текст с заменёнными PII-данными.
39 """
40 if not text:
41 return text
42
43 # Replace emails
44 text = re.sub(self.EMAIL_PATTERN, '[EMAIL]', text)
45
46 # Replace phones
47 text = re.sub(self.PHONE_PATTERN, '[PHONE]', text)
48
49 return text

51 def anonymize_structure(self, data: Dict | List | str) -> Dict | List | str:
52 """
53 @brief Рекурсивно анонимизирует PII в структуре данных.
54
55 Обходит dict и list рекурсивно, применяя anonymize_text() к строкам.
56
57 @param data Структура данных: dict, list или строка.
58 @return Та же структура с анонимизированными строками.
59 """
60 if isinstance(data, str):
61 return self.anonymize_text(data)
62 elif isinstance(data, dict):
63 return {k: self.anonymize_structure(v) for k, v in data.items()}
64 elif isinstance(data, list):
65 return [self.anonymize_structure(item) for item in data]
66 else:
67 return data
