54 @staticmethod
55 def select_model(query_or_complexity: str, prefer_privacy: bool = False, prefer_speed: bool = False) -> Union[str, Any]:
56 """
57 @brief Выбирает модель по уровню сложности или тексту запроса.
58
59 Если передан уровень сложности ("simple" / "medium" / "complex") —
60 возвращает строку с именем модели.
61 Если передан произвольный текст — классифицирует его и возвращает
62 провайдер (для обратной совместимости с тестами).
63
64 @param query_or_complexity Уровень сложности или текст запроса.
65 @param prefer_privacy Если True — принудительно использует vLLM (локальная модель).
66 @param prefer_speed Зарезервировано, пока не используется.
67 @return Имя модели (str) или экземпляр IModelProvider.
68 """
69 # Check if it's a complexity level
70 if query_or_complexity in ["simple", "medium", "complex"]:
71 return ModelSelector.MODELS.get(query_or_complexity, "gigachat_pro")
72
73 # If it's a query, use classifier and return provider
74 classifier = get_complexity_classifier()
75 result = classifier.classify(query_or_complexity)
76
77 model_name = result.suggested_model
78 if prefer_privacy:
79 model_name = "vllm"
80
81 return ModelFactory.get_model(model_name)
