83 async def create_graph_database(provider_type: str = "neo4j", force_new: bool = False) -> IGraphDatabase:
84 """
85 @brief Создаёт или возвращает singleton-экземпляр Neo4jProvider.
86
87 Выполняет health check при создании нового экземпляра.
88 При неудаче health check — закрывает провайдер и бросает RuntimeError.
89
90 @param provider_type Тип провайдера (только "neo4j").
91 @param force_new Если True — создаёт новый экземпляр, игнорируя кеш.
92 @return Экземпляр IGraphDatabase.
93 @throws RuntimeError Если Neo4j недоступен.
94 @throws ValueError Если передан неизвестный тип провайдера.
95 """
96 from app.core.config import settings
97
98 if provider_type != "neo4j":
99 raise ValueError(f"Unknown graph database provider: {provider_type}")
100
101 # Use singleton pattern unless force_new is True
102 if not force_new and DatabaseFactory._neo4j_instance is not None:
103 logger.info("Returning existing Neo4j provider instance (singleton)")
104 return DatabaseFactory._neo4j_instance
105
106 with DatabaseFactory._neo4j_lock:
107 # Double-check locking pattern
108 if not force_new and DatabaseFactory._neo4j_instance is not None:
109 return DatabaseFactory._neo4j_instance
110
111 logger.info("Creating new Neo4j provider instance...")
112 # Pass configuration
113 provider = Neo4jProvider(config=settings.DATABASE_CONFIG)
114
115 # Perform health check (fail-fast)
116 logger.info("Performing health check on Neo4j provider...")
117 health_ok = await provider.health_check()
118
119 if not health_ok:
120 logger.error("Neo4j health check failed! Database is unavailable.")
121 await provider.close()
122 raise RuntimeError(
123 "Neo4j database is unavailable. Please check your Neo4j connection settings and ensure the database is running."
124 )
125
126 logger.info("Neo4j health check passed successfully.")
127
128 # Cache the instance (singleton)
129 if not force_new:
130 DatabaseFactory._neo4j_instance = provider
131 logger.info("Neo4j provider instance cached for reuse.")
132
133 return provider
