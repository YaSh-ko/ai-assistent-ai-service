145async def get_rag_chain() -> RAGChain:
146 """
147 @brief DI-зависимость: возвращает singleton RAGChain с полной интеграцией.
148
149 При первом вызове инициализирует:
150 - DataAccessLayer (PostgreSQL + ChromaDB).
151 - BM25Provider + VectorSearchProvider → HybridSearchProvider.
152 - GraphRepository (Neo4j, опционально — деградирует gracefully).
153 - EmbeddingService, LLMService, ReasoningService, RerankerProvider, PIIService.
154
155 @return Полностью инициализированный RAGChain.
156 """
157 global _rag_chain
158 async with _rag_chain_lock:
159 if _rag_chain is None:
160 # 1. Database & DAL
161 dal = await get_dal_async()
162
163 # 2. Search Providers
164 # Postgres for BM25 (reuse pool from provider)
165 postgres_provider = DatabaseFactory.create_relational_database()
166 if not postgres_provider.pool:
167 await postgres_provider.connect()
168
169 # Vector store (Chroma)
170 vector_store = DatabaseFactory.create_vector_store("chroma")
171
172 # Instantiate search providers
173 bm25_provider = BM25Provider(postgres_provider)
174 vector_provider = VectorSearchProvider(vector_store)
175 hybrid_search_provider = HybridSearchProvider(bm25_provider, vector_provider)
176
177 # 3. Graph Repository (Neo4j)
178 graph_repository = None
179 try:
180 graph_db = await DatabaseFactory.create_graph_database()
181 graph_repository = GraphRepository(graph_db)
182 except Exception as e:
183 logger.warning(f"Neo4j unavailable, graph search disabled: {e}")
184
185 # 4. Services
186 chunking_service = ChunkingService()
187 embeddings_provider = GigaChatEmbeddings() # Real GigaChat embeddings
188
189 embedding_service = EmbeddingService(
190 chunking_service=chunking_service,
191 embeddings_provider=embeddings_provider,
192 embedding_repository=dal.embedding_repo
193 )
194
195 llm_service = await get_llm_service()
196 reasoning_service = await get_reasoning_service()
197 reranker_provider = RerankerProvider(llm_service)
198 pii_service = await get_pii_service()
199
200 # 5. Create RAGChain
201 _rag_chain = RAGChain(
202 dal=dal,
203 embedding_service=embedding_service,
204 llm_service=llm_service,
205 reasoning_service=reasoning_service,
206 hybrid_search_provider=hybrid_search_provider,
207 reranker_provider=reranker_provider,
208 graph_repository=graph_repository,
209 pii_service=pii_service
210 )
211
212 return _rag_chain
