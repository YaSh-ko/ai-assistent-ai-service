118 def _init_collection(self):
119 """Initialize or load the collection."""
120 Collection = self._pymilvus['Collection']
121 CollectionSchema = self._pymilvus['CollectionSchema']
122 FieldSchema = self._pymilvus['FieldSchema']
123 DataType = self._pymilvus['DataType']
124 utility = self._pymilvus['utility']
125
126 # Check if collection exists
127 if utility.has_collection(self._collection_name):
128 logger.info(f"Loading existing collection '{self._collection_name}'")
129 self._collection = Collection(self._collection_name)
130 self._collection.load()
131 else:
132 logger.info(f"Creating new collection '{self._collection_name}'")
133
134 # Define schema
135 fields = [
136 FieldSchema(name="id", dtype=DataType.VARCHAR, is_primary=True, max_length=256),
137 FieldSchema(name="embedding", dtype=DataType.FLOAT_VECTOR, dim=self._embedding_dim),
138 FieldSchema(name="page_content", dtype=DataType.VARCHAR, max_length=65535),
139 FieldSchema(name="metadata", dtype=DataType.JSON)
140 ]
141
142 schema = CollectionSchema(
143 fields=fields,
144 description="Chat history with embeddings"
145 )
146
147 # Create collection
148 self._collection = Collection(
149 name=self._collection_name,
150 schema=schema
151 )
152
153 # Create index for vector field
154 index_params = {
155 "metric_type": self._metric_type,
156 "index_type": "IVF_FLAT",
157 "params": {"nlist": 128}
158 }
159
160 self._collection.create_index(
161 field_name="embedding",
162 index_params=index_params
163 )
164
165 # Load collection into memory
166 self._collection.load()
167
168 logger.info(f"Collection '{self._collection_name}' created and loaded")

170 async def add_documents(self, documents: List[Dict[str, Any]], embeddings: List[List[float]]) -> None:
171 """
172 Add documents and their embeddings to Milvus.
173
174 Args:
175 documents: List of document dicts with 'id', 'page_content', 'metadata'
176 embeddings: List of embedding vectors
177 """
178 try:
179 if not documents or not embeddings:
180 logger.warning("No documents or embeddings to add")
181 return
182
183 if len(documents) != len(embeddings):
184 raise ValueError(f"Documents count ({len(documents)}) != embeddings count ({len(embeddings)})")
185
186 # Prepare data
187 ids = []
188 contents = []
189 metadatas = []
190
191 for i, doc in enumerate(documents):
192 # Generate ID if not provided
193 doc_id = doc.get("id")
194 if not doc_id:
195 import uuid
196 doc_id = str(uuid.uuid4())
197
198 ids.append(doc_id)
199 contents.append(doc.get("page_content", ""))
200 metadatas.append(doc.get("metadata", {}))
201
202 # Insert data
203 data = [
204 ids,
205 embeddings,
206 contents,
207 metadatas
208 ]
209
210 self._collection.insert(data)
211 self._collection.flush()
212
213 logger.info(f"Added {len(documents)} documents to Milvus")
214
215 except Exception as e:
216 logger.error(f"Error adding documents to Milvus: {e}")
217 raise

219 async def similarity_search(
220 self,
221 query_embedding: List[float],
222 k: int = 4,
223 filter: Optional[Dict[str, Any]] = None
224 ) -> List[Dict[str, Any]]:
225 """
226 Search for similar documents using vector similarity.
227
228 Args:
229 query_embedding: Query vector
230 k: Number of results to return
231 filter: Metadata filter (optional)
232
233 Returns:
234 List of documents with similarity scores
235 """
236 try:
237 # Prepare search parameters
238 search_params = {
239 "metric_type": self._metric_type,
240 "params": {"nprobe": 10}
241 }
242
243 # Build filter expression if provided
244 expr = None
245 if filter:
246 # Convert filter dict to Milvus expression
247 # Example: {"user_id": "123"} -> "metadata['user_id'] == '123'"
248 conditions = []
249 for key, value in filter.items():
250 if isinstance(value, bool):
251 conditions.append(f"metadata['{key}'] == {str(value).lower()}")
252 elif isinstance(value, str):
253 conditions.append(f"metadata['{key}'] == '{value}'")
254 elif isinstance(value, (int, float)):
255 conditions.append(f"metadata['{key}'] == {value}")
256
257 if conditions:
258 expr = " and ".join(conditions)
259
260 # Perform search
261 results = self._collection.search(
262 data=[query_embedding],
263 anns_field="embedding",
264 param=search_params,
265 limit=k,
266 expr=expr,
267 output_fields=["id", "page_content", "metadata"]
268 )
269
270 # Format results
271 documents = []
272 for hits in results:
273 for hit in hits:
274 documents.append({
275 "id": hit.entity.get("id"),
276 "page_content": hit.entity.get("page_content"),
277 "metadata": hit.entity.get("metadata", {}),
278 "score": float(hit.score)
279 })
280
281 logger.info(f"Found {len(documents)} similar documents")
282 return documents
283
284 except Exception as e:
285 logger.error(f"Error during similarity search in Milvus: {e}")
286 raise
