141 async def save_thread_state(
142 self, thread_id: str, run_id: str, assistant_id: Optional[str],
143 final_messages: list, parent_checkpoint_id: Optional[str] = None
144 ) -> None:
145 """
146 @brief Сохраняет результат run как checkpoint-состояние сессии.
147
148 Используется LangGraph SDK для навигации по веткам диалога.
149 Новый checkpoint добавляется в начало списка states.
150
151 @param thread_id UUID треда.
152 @param run_id UUID текущего run.
153 @param assistant_id ID ассистента.
154 @param final_messages Финальный список сообщений диалога.
155 @param parent_checkpoint_id ID родительского checkpoint (опционально).
156 """
157 import uuid
158 from datetime import datetime, timezone
159
160 session = await self.get_session(thread_id)
161 if not session:
162 return
163
164 checkpoint_id = str(uuid.uuid4())
165
166 if parent_checkpoint_id:
167 parent_cp = {"thread_id": thread_id, "checkpoint_ns": "", "checkpoint_id": parent_checkpoint_id}
168 elif session.states:
169 parent_cp = session.states[0]["checkpoint"]
170 else:
171 parent_cp = None
172
173 history_entry = {
174 "values": {"messages": final_messages},
175 "next": [],
176 "checkpoint": {
177 "thread_id": thread_id,
178 "checkpoint_ns": "",
179 "checkpoint_id": checkpoint_id,
180 },
181 "parent_checkpoint": parent_cp,
182 "metadata": {
183 "run_id": run_id,
184 "assistant_id": assistant_id,
185 },
186 "created_at": datetime.now(timezone.utc).isoformat(),
187 "tasks": [],
188 }
189
190 new_states = [history_entry] + (session.states or [])
191 await self.update_session(thread_id, {
192 "states": new_states,
193 "history": final_messages # Sync flat history with the latest state
194 })

196 async def get_messages_by_checkpoint(self, thread_id: str, checkpoint_id: str) -> list:
197 """
198 @brief Возвращает сообщения для конкретного checkpoint.
199
200 @param thread_id UUID треда.
201 @param checkpoint_id UUID checkpoint.
202 @return Список сообщений или текущая история, если checkpoint не найден.
203 """
204 session = await self.get_session(thread_id)
205 if not session or not session.states:
206 return []
207
208 for state in session.states:
209 cp = state.get("checkpoint", {})
210 if cp.get("checkpoint_id") == checkpoint_id:
211 return state.get("values", {}).get("messages", [])
212
213 # Fallback to current history if not found in states
