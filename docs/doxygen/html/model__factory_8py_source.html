53 @classmethod
54 def get_model(cls, model_name: Optional[str] = None) -> IModelProvider:
55 """
56 @brief Возвращает провайдер модели по имени.
57
58 Если провайдер уже создан — возвращает из кеша.
59 Иначе создаёт новый и кеширует.
60
61 @param model_name Имя модели: vllm, gigachat, gigachat_pro, gigachat_max.
62 Если None — используется текущая модель из настроек.
63 @return Экземпляр IModelProvider.
64 @throws ValueError Если модель не найдена.
65 """
66 if model_name is None:
67 model_name = cls._current_model or settings.CURRENT_MODEL
68
69 model_name = model_name.lower().replace("-", "_")
70
71 with cls._lock:
72 # Проверяем кеш
73 if model_name in cls._providers_cache:
74 logger.debug(f"Returning cached provider for {model_name}")
75 return cls._providers_cache[model_name]
76
77 # Создаём новый провайдер
78 provider = cls._create_provider(model_name)
79
80 # Кешируем
81 cls._providers_cache[model_name] = provider
82
83 return provider

128 @classmethod
129 async def get_model_with_fallback(
130 cls,
131 model_name: Optional[str] = None
132 ) -> IModelProvider:
133 """
134 @brief Возвращает первый доступный провайдер с автоматическим fallback.
135
136 Проверяет доступность каждой модели в порядке fallback_order,
137 начиная с предпочтительной.
138
139 @param model_name Предпочтительная модель (опционально).
140 @return Первый доступный IModelProvider.
141 @throws ModelUnavailableError Если все модели недоступны.
142 """
143 preferred_model = model_name or settings.CURRENT_MODEL
144
145 # Строим порядок для fallback, начиная с предпочтительной модели
146 fallback_order = [preferred_model]
147 for model in cls._fallback_order:
148 if model not in fallback_order:
149 fallback_order.append(model)
150
151 # Пробуем каждую модель
152 for candidate in fallback_order:
153 try:
154 provider = cls.get_model(candidate)
155
156 if await provider.is_available():
157 if candidate != preferred_model:
158 logger.warning(
159 f"Using fallback model {candidate} instead of {preferred_model}"
160 )
161 return provider
162 else:
163 logger.warning(f"Model {candidate} is not available")
164
165 except Exception as e:
166 logger.warning(f"Failed to get model {candidate}: {e}")
167 continue
168
169 raise ModelUnavailableError(
170 "all_models",
171 "All models are unavailable"
172 )
