52 async def _perform_reasoning(
53 self,
54 query: str,
55 context: Optional[Dict[str, Any]],
56 **kwargs
57 ) -> Any:
58 """
59 Основной метод выполнения рассуждения, оркестрирующий 4 шага.
60
61 Args:
62 query: Вопрос пользователя.
63 context: Контекст (результаты поиска и т.д.).
64
65 Returns:
66 Финальный синтезированный ответ.
67 """
68 step_durations = {}
69
70 try:
71 # Step 1: Understand
72 t0 = time.time()
73 understanding = await asyncio.wait_for(
74 self.understand(query, context),
75 timeout=self.timeout_per_step
76 )
77 step_durations["understand"] = (time.time() - t0) * 1000
78
79 # Check for empty context edge case
80 if not context and not understanding.get("can_answer_without_context", False):
81 # If context is empty and model thinks it needs it, we might want to stop or warn
82 logger.warning("Context is empty, reasoning might be limited.")
83
84 # Step 2: Plan
85 t0 = time.time()
86 plan = await asyncio.wait_for(
87 self.plan(understanding),
88 timeout=self.timeout_per_step
89 )
90 step_durations["plan"] = (time.time() - t0) * 1000
91
92 # Step 3: Execute
93 t0 = time.time()
94 execution_results = await asyncio.wait_for(
95 self.execute(plan),
96 timeout=self.timeout_per_step
97 )
98 step_durations["execute"] = (time.time() - t0) * 1000
99
100 # Step 4: Verify
101 if self.enable_verification:
102 t0 = time.time()
103 final_result = await asyncio.wait_for(
104 self.verify(execution_results, query),
105 timeout=self.timeout_per_step
106 )
107 step_durations["verify"] = (time.time() - t0) * 1000
108 else:
109 # If verification is disabled, just return execution results formatted
110 final_result = str(execution_results)
111 step_durations["verify"] = 0
112
113 # Record metrics
114 self.metrics.record_execution(
115 success=True,
116 duration_ms=sum(step_durations.values()),
117 step_durations=step_durations
118 )
119
120 return final_result
121
122 except asyncio.TimeoutError as e:
123 logger.error(f"Reasoning step timed out: {str(e)}")
124 self.metrics.record_execution(success=False, duration_ms=0, step_durations=step_durations)
125 raise TimeoutError("Reasoning step timed out")
126 except Exception as e:
127 logger.error(f"Reasoning failed: {str(e)}")
128 self.metrics.record_execution(success=False, duration_ms=0, step_durations=step_durations)
129 raise e
