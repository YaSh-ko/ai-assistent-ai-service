69 query = f"""

156 async def graph_search(
157 self,
158 user_id: str,
159 query_concepts: List[str],
160 limit: int = 5
161 ) -> List[Dict[str, Any]]:
162 """
163 Search for entries connected to given concepts.
164
165 Args:
166 user_id: User ID to filter by.
167 query_concepts: List of concept names to search for.
168 limit: Maximum number of results.
169
170 Returns:
171 List of entries ordered by relevance (number of matching concepts).
172 """
173 if not query_concepts:
174 return []
175
176 # Cypher query to find entries connected to concepts
177 query = """
178 MATCH (e:Entry {user_id: $user_id})-[r:HAS_CONCEPT|RELATES_TO]->(c:Concept)
179 WHERE c.name IN $concepts
180 WITH e, COUNT(DISTINCT c) as concept_matches, COLLECT(c.name) as matched_concepts
181 RETURN e, concept_matches, matched_concepts
182 ORDER BY concept_matches DESC
183 LIMIT $limit
184 """
185
186 results = await self.db.execute_read(query, {
187 "user_id": user_id,
188 "concepts": query_concepts,
189 "limit": limit
190 })
191
192 # Format results
193 formatted = []
194 for record in results:
195 entry_data = record.get('e', {})
196 formatted.append({
197 "id": entry_data.get('id'),
198 "title": entry_data.get('title'),
199 "description": entry_data.get('description'),
200 "event_date": entry_data.get('event_date'),
201 "graph_score": record.get('concept_matches', 0),
202 "matched_concepts": record.get('matched_concepts', []),
203 "source": "graph"
204 })
205
206 return formatted

208 async def find_related_entries(
209 self,
210 entry_ids: List[str],
211 depth: int = 2,
212 limit: int = 10
213 ) -> List[Dict[str, Any]]:
214 """
215 Find entries related to given entries through graph traversal.
216
217 Args:
218 entry_ids: List of entry IDs to start from.
219 depth: Maximum traversal depth.
220 limit: Maximum number of results.
221
222 Returns:
223 List of related entries with relationship info.
224 """
225 if not entry_ids:
226 return []
227
228 # Cypher query to find related entries through multiple hops
229 query = f"""
230 MATCH (e1:Entry)-[r*1..{depth}]-(e2:Entry)
231 WHERE e1.id IN $entry_ids AND e2.id NOT IN $entry_ids
232 WITH e2, COUNT(*) as path_count, COLLECT(DISTINCT type(r[0])) as relationship_types
233 RETURN e2, path_count, relationship_types
234 ORDER BY path_count DESC
235 LIMIT $limit
236 """
237
238 results = await self.db.execute_read(query, {
239 "entry_ids": entry_ids,
240 "limit": limit
241 })
242
243 formatted = []
244 for record in results:
245 entry_data = record.get('e2', {})
246 formatted.append({
247 "id": entry_data.get('id'),
248 "title": entry_data.get('title'),
249 "description": entry_data.get('description'),
250 "graph_score": record.get('path_count', 0),
251 "relationship_types": record.get('relationship_types', []),
252 "source": "graph_related"
253 })
254
255 return formatted
