262 raise ModelError(

269 line = line.decode("utf-8").strip()
270
271 if not line:
272 continue

274 if line.startswith("data:"):
275 data = line[5:].strip()
276
277 if data == "[DONE]":
278 yield StreamChunk(
279 content="",
280 is_final=True,
281 model_name=self.name,
282 finish_reason="stop"

287 chunk_data = json.loads(data)
288 choices = chunk_data.get("choices", [])
289

291 delta = choices[0].get("delta", {})
292 content = delta.get("content", "")
293 finish_reason = choices[0].get("finish_reason")
294
295 if content:
296 total_content += content
297 yield StreamChunk(
298 content=content,
299 is_final=False,
300 model_name=self.name
301 )
302
303 if finish_reason:
304 yield StreamChunk(
305 content="",
306 is_final=True,
307 model_name=self.name,
308 finish_reason=finish_reason
309 )
310 break
311
312 except json.JSONDecodeError:
313 logger.warning(f"Failed to parse SSE chunk: {data}")
314 continue
315
316 # Запись метрик
317 latency = self._measure_latency(start_time)
318 self._metrics.record_request(
319 model_name=self.name,
320 success=True,
321 latency_ms=latency,
322 tokens=len(total_content) // 4
323 )
324

326
