38def _format_dialogue(messages: List[dict]) -> str:
39 """Форматирует последние сообщения для вставки в промпт."""
40 lines = []
41 for msg in messages:
42 role = msg.get("type", msg.get("role", ""))
43 content = msg.get("content", "")
44 if isinstance(content, list):
45 content = " ".join(
46 block.get("text", "") if isinstance(block, dict) else str(block)
47 for block in content
48 )
49 if role in ("human", "user"):
50 lines.append(f"Пользователь: {content}")
51 elif role in ("ai", "assistant"):
52 lines.append(f"Delёz: {content}")
53 return "\n".join(lines)

56def _parse_hints(raw: str) -> List[str]:
57 """Извлекает ровно 3 строки из ответа LLM."""
58 lines = [line.strip().lstrip("-•*0123456789.) ") for line in raw.strip().splitlines()]
59 lines = [l for l in lines if l]
60 if len(lines) >= 3:
61 return lines[:3]
62 # Если LLM вернул меньше строк, дополняем fallback-подсказками
63 while len(lines) < 3:
64 lines.append(_FALLBACK_HINTS[len(lines)])
65 return lines
