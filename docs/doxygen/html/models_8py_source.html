89@router.post("/select", summary="Выбор модели")
90async def select_model(request: SelectModelRequest):
91 """
92 Выбрать модель по имени.
93 Модель станет текущей по умолчанию.
94 """
95 model_name = request.model_name.lower()
96
97 if model_name not in ModelFactory.get_available_models():
98 raise HTTPException(
99 status_code=400,
100 detail=f"Unknown model: {model_name}. Available: {ModelFactory.get_available_models()}"
101 )
102
103 # Проверяем доступность
104 provider = ModelFactory.get_model(model_name)
105 is_available = await provider.is_available()
106
107 if not is_available:
108 raise HTTPException(
109 status_code=503,
110 detail=f"Model {model_name} is not available"
111 )
112
113 # Устанавливаем как текущую
114 ModelFactory.set_current_model(model_name)
115
116 logger.info(f"Model switched to: {model_name}")
117
118 return {
119 "message": f"Model switched to {model_name}",
120 "current_model": model_name
121 }
