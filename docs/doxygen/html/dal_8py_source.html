70 """
71 @brief Атомарно сохраняет запись дневника и её эмбеддинг.
72
73 Шаги:
74 1. Создаёт запись в PostgreSQL через entry_repo.
75 2. Сохраняет эмбеддинг в ChromaDB (если не передан — использует нулевой вектор).
76 3. Привязывает запись к диалогу через entry_thread_repo (если thread_id задан).
77
78 @param user_id ID пользователя-владельца.
79 @param title Заголовок записи.
80 @param description Текст записи.
81 @param event_date Дата события.
82 @param thread_id ID диалога для привязки (опционально).
83 @param embedding Вектор эмбеддинга (опционально, 1024-мерный).
84 @return Объект записи с атрибутом id (SimpleNamespace).
85 @throws RuntimeError Если запись не удалось создать в PostgreSQL.
86 """
87 # 1. Save to PostgreSQL
88 entry_dict = await self.entry_repo.create(user_id, title, description, event_date)
89 if not entry_dict:
90 raise RuntimeError("Failed to create entry in PostgreSQL")
91
92 # 2. Create Embedding
93 if embedding is None:
94 # Use dummy embedding if not provided (fallback)
95 embedding = [0.0] * 1024
96
97 metadata = {
98 "entry_id": str(entry_dict['id']),
99 "user_id": user_id,
100 "event_date": str(event_date)
101 }
102 if thread_id:
103 metadata["thread_id"] = thread_id
104
105 page_content = f"{title}\n{description}"
106
107 await self.embedding_repo.add_embedding(
108 document_id=str(entry_dict['id']),
109 embedding=embedding,
110 metadata=metadata,
111 page_content=page_content
112 )
113
114 # 3. Link to thread if provided
115 if thread_id:
116 await self.entry_thread_repo.create(entry_dict['id'], thread_id, "BELONGS_TO", user_id)
117
118 # Return as object to satisfy test expectation (entry.id)
119 return SimpleNamespace(**entry_dict)
