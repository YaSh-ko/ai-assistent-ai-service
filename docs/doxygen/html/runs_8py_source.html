61def _extract_text_content(content) -> str:
62 """Extract text from message content (string or list of blocks)."""
63 if isinstance(content, str):
64 return content
65 if isinstance(content, list):
66 text_parts = []
67 for block in content:
68 if isinstance(block, str):
69 text_parts.append(block)
70 elif isinstance(block, dict) and block.get("type") == "text":
71 text_parts.append(block.get("text", ""))
72 return " ".join(text_parts)
73 return ""

77def _extract_user_message(input_data: Optional[Dict[str, Any]]) -> Optional[str]:
78 """Extract user message text from the SDK input format."""
79 if not input_data:
80 return None
81
82 messages = input_data.get("messages", [])
83 if not messages:
84 return None
85
86 # Find the last human message
87 for msg in reversed(messages):
88 if isinstance(msg, dict) and msg.get("type") == "human":
89 return _extract_text_content(msg.get("content", ""))
90 return None

93def _trim_to_last_human(messages: list) -> list:
94 """Trim messages list to end at the last human message (inclusive)."""
95 last_human_idx = None
96 for i, msg in enumerate(messages):
97 if isinstance(msg, dict) and msg.get("type") == "human":
98 last_human_idx = i
99 if last_human_idx is None:
100 return messages
101 return messages[: last_human_idx + 1]

123async def _handle_regeneration_flow(
124 thread_id: str,
125 checkpoint_id: str,
126 existing_messages: list,
127 session_manager: SessionManager
128) -> tuple[Optional[str], list]:
129 """Handle the logic for replaying from a checkpoint."""
130 base_messages = await session_manager.get_messages_by_checkpoint(thread_id, checkpoint_id)
131 if not base_messages:
132 base_messages = existing_messages
133
134 full_messages_history = _trim_to_last_human(base_messages)
135 user_message = None
136
137 # Extract the question to re-ask
138 for msg in reversed(full_messages_history):
139 if isinstance(msg, dict) and msg.get("type") == "human":
140 user_message = _extract_text_content(msg.get("content", ""))
141 break
142
143 return user_message, full_messages_history

146def _merge_normal_messages(existing_messages: list, input_data: Optional[Dict[str, Any]]) -> list:
147 """Merge incoming messages with existing history."""
148 new_input_messages = input_data.get("messages", []) if input_data else []
149 existing_ids = {m.get("id") for m in existing_messages if isinstance(m, dict) and m.get("id")}
150 filtered_new_messages = [
151 m for m in new_input_messages
152 if not (isinstance(m, dict) and m.get("id") in existing_ids)
153 ]
154 return existing_messages + filtered_new_messages

157async def _stream_and_save_response(
158 thread_id: str,
159 run_id: str,
160 assistant_id: Optional[str],
161 user_message: str,
162 user_id: str,
163 full_messages_history: list,
164 checkpoint_id: Optional[str],
165 is_regeneration: bool,
166 rag_chain: Any,
167 session_manager: SessionManager
168):
169 """Internal generator to stream RAGChain output and save state."""
170 yield _format_sse("metadata", {"run_id": run_id})
171 yield _format_sse("values", {"messages": full_messages_history})
172
173 try:
174 llm_stream, _ = await rag_chain.process_user_message(
175 thread_id=thread_id,
176 user_question=user_message,
177 user_id=user_id,
178 )
179
180 full_response = ""
181 ai_message_id = str(uuid.uuid4())
182 async for chunk in llm_stream:
183 if chunk.content:
184 full_response += chunk.content
185 yield _format_sse("values", {
186 "messages": full_messages_history + [{"type": "ai", "id": ai_message_id, "content": full_response}]
187 })
188
189 if not full_response:
190 full_response = "Извините, не удалось сгенерировать ответ."
191
192 await session_manager.save_thread_state(
193 thread_id, run_id, assistant_id,
194 full_messages_history + [{"type": "ai", "id": ai_message_id, "content": full_response}],
195 parent_checkpoint_id=checkpoint_id if is_regeneration else None
196 )
197 except Exception as e:
198 logger.error(f"Error in _stream_and_save_response: {e}", exc_info=True)
199 yield _format_sse("error", {"message": str(e), "name": "StreamError"})

207 session_manager: Annotated[SessionManager, Depends(get_session_manager)],

210 """
211 Create a run and stream results (LangGraph SDK compatible).
212
213 This is the main endpoint called by agent-chat-ui's useStream() hook.
214 POST /threads/{thread_id}/runs/stream
215 """
216 user_id = x_user_id or "default_user"
217 logger.info(f"Stream run for thread {thread_id}, user {user_id}")
218
219 existing_messages = await _get_existing_messages(thread_id, session_manager)
220 user_message = _extract_user_message(request.input)
221 is_regeneration = False
222 full_messages_history = []
223 checkpoint_id = _parse_checkpoint_id(request)
224
225 if not user_message and checkpoint_id:
226 is_regeneration = True
227 user_message, full_messages_history = await _handle_regeneration_flow(
228 thread_id, checkpoint_id, existing_messages, session_manager
229 )
230
231 if not user_message:
232 async def empty_response():
233 yield _format_sse("metadata", {"run_id": str(uuid.uuid4())})
234 yield _format_sse("values", {"messages": existing_messages})
235
236 return StreamingResponse(
237 empty_response(),
238 media_type="text/event-stream",
239 headers={"Cache-Control": "no-cache", "Connection": "keep-alive"},
240 )
241
242 if not is_regeneration:
243 full_messages_history = _merge_normal_messages(existing_messages, request.input)
244
245 run_id = str(uuid.uuid4())
246 return StreamingResponse(

248 thread_id, run_id, request.assistant_id, user_message, user_id,
