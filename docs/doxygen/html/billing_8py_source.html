57 def get_pricing(self, model: str) -> dict[str, float]:
58 """
59 @brief Возвращает тариф для указанной модели.
60
61 Использует prefix-matching: "GigaChat-Pro" → PRICING["GigaChat-Pro"].
62 При неизвестной модели возвращает тариф "default" (GigaChat Base).
63
64 @param model Имя модели (например, "GigaChat-Pro").
65 @return Словарь с ключами "input" и "output" (руб / 1000 токенов).
66 """
67 for key in PRICING:
68 if key != "default" and model.startswith(key):
69 return PRICING[key]

72 def calculate_cost(
73 self,
74 model: str,
75 input_tokens: int,
76 output_tokens: int,
77 ) -> float:
78 """
79 @brief Рассчитывает стоимость запроса в рублях.
80
81 @param model Имя модели.
82 @param input_tokens Количество входных токенов (промпт).
83 @param output_tokens Количество выходных токенов (ответ).
84 @return Стоимость в рублях, округлённая до 4 знаков.
85 """
86 pricing = self.get_pricing(model)
87 cost = (input_tokens * pricing["input"] + output_tokens * pricing["output"]) / 1000
88 return round(cost, 4)

90 def record(
91 self,
92 model: str,
93 input_tokens: int,
94 output_tokens: int,
95 ) -> float:
96 """
97 @brief Записывает токены и стоимость в Prometheus-метрики.
98
99 Инкрементирует ai_tokens_total (input/output) и ai_cost_total_rub.
100 Вызывается автоматически из LLMService после каждой генерации.
101
102 @param model Имя модели.

104 @param output_tokens Количество выходных токенов.

106 """
107 cost = self.calculate_cost(model, input_tokens, output_tokens)
108 ai_tokens_total.labels(model=model, token_type="input").inc(input_tokens)
