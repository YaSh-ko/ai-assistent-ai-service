157 def build_graph(self) -> StateGraph:
158 """
159 @brief Строит и компилирует LangGraph-граф пайплайна.
160
161 Определяет узлы, рёбра и условную маршрутизацию
162 (simple → generate_response, complex → cot_reasoning).
163
164 @return Скомпилированный граф (CompiledGraph).
165 """
166 workflow = StateGraph(RAGState)
167
168 # Определение узлов
169 workflow.add_node("classify_query", self.classify_query)
170 workflow.add_node("retrieve_events", self.retrieve_events)
171 workflow.add_node("filter_relevant", self.filter_relevant)
172 workflow.add_node("cot_reasoning", self.cot_reasoning)
173 workflow.add_node("generate_response", self.generate_response)
174 workflow.add_node("save_to_db", self.save_to_db)
175
176 # Определение связей
177 workflow.set_entry_point("classify_query")
178 workflow.add_edge("classify_query", "retrieve_events")
179 workflow.add_edge("retrieve_events", "filter_relevant")
180
181 # Conditional routing based on complexity
182 workflow.add_conditional_edges(
183 "filter_relevant",
184 self._route_based_on_complexity,
185 {
186 "simple": "generate_response",
187 "complex": "cot_reasoning"
188 }
189 )
190
191 workflow.add_edge("cot_reasoning", "generate_response")
192 workflow.add_edge("generate_response", "save_to_db")
193 workflow.add_edge("save_to_db", END)
194
195 return workflow.compile()

247 async def retrieve_events(self, state: RAGState) -> RAGState:
248 """
249 @brief Шаг 1: гибридный поиск похожих записей.
250
251 Параллельно выполняет:
252 - HybridSearchProvider (BM25 + Vector) → топ-10.
253 - GraphRepository.graph_search() по концептам из запроса.
254
255 Результаты дедуплицируются по id и сортируются по суммарному score.
256
257 @param state Текущее состояние пайплайна.
258 @return Обновлённое состояние с заполненными search_results и graph_insights.
259 """
260 question = state["question"]
261 user_id = state.get("user_id")
262
263 logger.info(f"[ШАГ 1] Поиск событий для: '{question[:50]}...'")
264
265 try:
266 # Генерируем embedding для запроса
267 if self.embedding_service:
268 query_embedding = await self.embedding_service.generate_embedding(question)
269 state["query_embedding"] = query_embedding
270 else:
271 query_embedding = [0.0] * 1024 # Заглушка
272 state["query_embedding"] = query_embedding
273
274 # Extract concepts from question for graph search
275 # Simple: split question into words, filter stopwords, use as concepts
276 # In production, use NER or LLM to extract concepts
277 query_concepts = [w.lower() for w in question.split() if len(w) > 3][:5]
278
279 # Run searches in parallel
280 search_tasks = []
281
282 # Hybrid (BM25 + Vector)
283 if self.hybrid_search_provider:
284 search_tasks.append(
285 self.hybrid_search_provider.search(
286 query=question,
287 query_embedding=query_embedding,
288 top_k=self.search_top_k,
289 user_id=user_id
290 )
291 )
292 else:
293 search_tasks.append(
294 self._fallback_search(query_embedding, user_id)
295 )
296
297 # Graph search (if available)
298 if self.graph_repository:
299 search_tasks.append(
300 self.graph_repository.graph_search(
301 user_id=user_id,
302 query_concepts=query_concepts,
303 limit=self.search_top_k
304 )
305 )
306 else:
307 search_tasks.append(self._get_empty_list_async())
308
309 # Execute all searches in parallel
310 results = await asyncio.gather(*search_tasks, return_exceptions=True)
311
312 hybrid_results = results[0] if not isinstance(results[0], Exception) else []
313 graph_results = results[1] if len(results) > 1 and not isinstance(results[1], Exception) else []
314
315 if isinstance(results[0], Exception):
316 logger.warning(f"Hybrid search failed: {results[0]}")
317 if len(results) > 1 and isinstance(results[1], Exception):
318 logger.warning(f"Graph search failed: {results[1]}")
319
320 # Merge results (deduplicate by id)
321 merged = {}
322 for r in hybrid_results:
323 doc_id = str(r.get('id', ''))
324 if doc_id:
325 merged[doc_id] = r
326
327 for r in graph_results:
328 doc_id = str(r.get('id', ''))
329 if doc_id and doc_id not in merged:
330 merged[doc_id] = r
331 elif doc_id and doc_id in merged:
332 # Combine scores
333 merged[doc_id]['graph_score'] = r.get('graph_score', 0)
334 merged[doc_id]['matched_concepts'] = r.get('matched_concepts', [])
335
336 # Sort by combined score
337 search_results = list(merged.values())
338 search_results.sort(
339 key=lambda x: (x.get('final_score', 0) + x.get('graph_score', 0) * 0.3),
340 reverse=True
341 )
342
343 state["search_results"] = search_results[:self.search_top_k]
344 state["graph_insights"] = [{"type": "graph_search", "concepts": query_concepts}]
345
346 logger.info(f"[ШАГ 1] Найдено {len(state['search_results'])} событий (hybrid: {len(hybrid_results)}, graph: {len(graph_results)})")
347
348 except Exception as e:
349 logger.error(f"[ШАГ 1] Ошибка поиска: {e}")
350 state["search_results"] = []
351
352 return state

392 async def filter_relevant(self, state: RAGState) -> RAGState:
393 """
394 @brief Шаг 2: фильтрация результатов через Reranker.
395
396 Оставляет топ-5 наиболее релевантных записей.
397 Если Reranker недоступен — берёт топ по final_score.
398 Формирует текстовый контекст для LLM.
399
400 @param state Текущее состояние пайплайна.
401 @return Обновлённое состояние с filtered_results и context.
402 """
403 search_results = state.get("search_results", [])
404 question = state["question"]
405
406 logger.info(f"[ШАГ 2] Фильтрация {len(search_results)} результатов")
407
408 if not search_results:
409 state["filtered_results"] = []
410 state["context"] = ""
411 return state
412
413 try:
414 if self.reranker_provider:
415 # Используем reranker для переранжирования
416 filtered_results = await self.reranker_provider.rerank(
417 query=question,
418 documents=search_results,
419 top_k=self.rerank_top_k
420 )
421 else:
422 # Fallback: берём топ по final_score
423 sorted_results = sorted(
424 search_results,
425 key=lambda x: x.get('final_score', x.get('score', 0)),
426 reverse=True
427 )
428 filtered_results = sorted_results[:self.rerank_top_k]
429
430 state["filtered_results"] = filtered_results
431
432 # Формируем контекст для LLM
433 context = self._build_context_from_results(filtered_results)
434 state["context"] = context
435
436 logger.info(f"[ШАГ 2] Отфильтровано до {len(filtered_results)} релевантных событий")
437
438 except Exception as e:
439 logger.error(f"[ШАГ 2] Ошибка фильтрации: {e}")
440 state["filtered_results"] = search_results[:self.rerank_top_k]
441 state["context"] = self._build_context_from_results(state["filtered_results"])
442
443 return state

445 def _build_context_from_results(self, results: List[Dict[str, Any]]) -> str:
446 """
447 @brief Формирует текстовый контекст из результатов поиска.
448
449 Каждая запись форматируется с датой, заголовком, содержанием и score.
450
451 @param results Список документов из поиска.
452 @return Многострочная строка с контекстом для LLM.
453 """
454 if not results:
455 return ""
456
457 context_parts = ["Релевантные записи из дневника пользователя:\n"]
458
459 for i, result in enumerate(results, 1):
460 title = result.get('title', '')
461 description = result.get('description', result.get('page_content', ''))
462 event_date = result.get('event_date', '')
463 score = result.get('final_score', result.get('score', 0))
464
465 context_parts.append(f"""
466--- Запись {i} (релевантность: {score:.2f}) ---
467Дата: {event_date}
468Заголовок: {title}
469Содержание: {description}
470""")
471
472 return "\n".join(context_parts)

478 async def cot_reasoning(self, state: RAGState) -> RAGState:
479 """
480 @brief Шаг 3: Chain-of-Thought рассуждение (только для COMPLEX).
481
482 Передаёт вопрос и контекст в ReasoningService, который выбирает
483 движок (CoT или Reflection) и возвращает шаги рассуждения.
484
485 @param state Текущее состояние пайплайна.
486 @return Обновлённое состояние с reasoning_steps и reasoning_metadata.
487 """
488 question = state["question"]
489 filtered_results = state.get("filtered_results", [])
490 user_id = state.get("user_id")
491 complexity = state.get("complexity", "simple")
492
493 logger.info(f"[ШАГ 3] CoT Reasoning для: '{question[:50]}...' (complexity: {complexity})")
494
495 try:
496 # Подготовка контекста для reasoning
497 context = {
498 "rag_results": filtered_results,
499 "user_id": user_id,
500 "thread_id": state.get("thread_id")
501 }
502
503 # Выполнение reasoning через сервис
504 result = await self.reasoning_service.execute_reasoning(
505 question=question,
506 context=context,
507 task_type=complexity, # Используем сложность как тип задачи
508 user_id=user_id
509 )
510
511 # Обновление состояния
512 state["reasoning_steps"] = result.get("steps", [])
513 state["reasoning_engine_used"] = result.get("metadata", {}).get("type", "unknown")
514 state["reasoning_metadata"] = result.get("metadata", {})
515
516 # Если есть ответ от reasoning (например, финальный), можно его использовать
517 # Но обычно мы передаем шаги в generate_response для финальной генерации в нужном стиле
518
519 logger.info(f"[ШАГ 3] Завершено: {len(state['reasoning_steps'])} шагов рассуждения")
520
521 except Exception as e:
522 logger.error(f"[ШАГ 3] Ошибка CoT Reasoning: {e}")
523 state["reasoning_steps"] = []
524 state["reasoning_engine_used"] = "failed"
525 state["reasoning_metadata"] = {"error": str(e)}
526
527 return state

569 async def generate_response(self, state: RAGState) -> RAGState:
570 """
571 @brief Шаг 4: генерация финального ответа через GigaChat.
572
573 Формирует промпт из истории диалога, RAG-контекста и шагов CoT.
574 Выбирает модель и параметры через ModelSelector.
575 Сохраняет ответ в state["answer"].
576
577 @param state Текущее состояние пайплайна.
578 @return Обновлённое состояние с заполненным answer и selected_model.
579 """
580 from app.core.model_selector import ModelSelector
581
582 question = state["question"]
583 context = state.get("context", "")
584 thread_id = state.get("thread_id")
585 complexity = state.get("complexity", "simple")
586 reasoning_steps = state.get("reasoning_steps", [])
587
588 # Select model and params
589 selected_model = ModelSelector.select_model(complexity)
590 params = ModelSelector.get_params(complexity)
591
592 state["selected_model"] = selected_model
593 logger.info(f"[ШАГ 4] Генерация ответа с {selected_model} (params: {params})")
594

634 def _build_full_prompt(
635 self,
636 question: str,
637 context: str,
638 reasoning_steps: List[Dict],
639 history: List[Dict] = None
640 ) -> str:
641 """
642 @brief Собирает полный промпт для LLM.
643
644 Структура промпта:
645 1. История диалога (последние 5 сообщений).
646 2. RAG-контекст из записей дневника.
647 3. Шаги CoT-рассуждения (до 5 шагов, только для COMPLEX).
648 4. Текущий вопрос пользователя.
649
650 @param question Текущий вопрос.
651 @param context Текстовый контекст из RAG.
652 @param reasoning_steps Шаги CoT-рассуждения.
653 @param history История диалога (опционально).
654 @return Готовый промпт в виде строки.
655 """
656 prompt_parts = []
657
658 # Добавляем историю диалога
659 if history:
660 prompt_parts.append("ИСТОРИЯ ДИАЛОГА (сессия):")
661 for msg in history:
662 role = msg.get('role', 'unknown')
663 content = msg.get('content', '')
664 prompt_parts.append(f"{role.capitalize()}: {content}")
665 prompt_parts.append("\n─────────────────────────────────────────\n")
666
667 # Добавляем контекст из RAG
668 if context:
669 prompt_parts.append("RAG КОНТЕКСТ (события из истории):")
670 prompt_parts.append(context)
671 prompt_parts.append("\n─────────────────────────────────────────\n")
672
673 # Добавляем инсайты из рассуждений (для сложных запросов)
674 if reasoning_steps:
675 prompt_parts.append("REASONING КОНТЕКСТ:")
676 prompt_parts.append("Вот цепочка рассуждений:")
677 # Limit to top 5 steps to save tokens
678 for step in reasoning_steps[:5]:
679 desc = step.get('description', '')
680 thought = step.get('thought', '')
681 obs = step.get('observation', '')
682 prompt_parts.append(f"- Шаг {step.get('step_number', '?')}: {desc}")
683 if thought:
684 prompt_parts.append(f" Мысль: {thought}")
685 if obs:
686 prompt_parts.append(f" Результат: {obs}")
687 prompt_parts.append("\n─────────────────────────────────────────\n")
688
689 # Добавляем вопрос пользователя
690 prompt_parts.append("USER (текущий вопрос):")
691 prompt_parts.append(f"\"{question}\"")
692
693 return "\n".join(prompt_parts)

766 def _parse_event_date(self, date_val: Any) -> date:
767 """
768 @brief Парсит дату события из различных форматов.
769
770 Поддерживает: date, datetime, ISO-строку, "%Y-%m-%d", "%d.%m.%Y".
771 При неудаче возвращает сегодняшнюю дату.
772
773 @param date_val Значение даты в любом из поддерживаемых форматов.
774 @return Объект datetime.date.
775 """
776 if not date_val:
777 return date.today()
778
779 if isinstance(date_val, date):
780 return date_val
781
782 if isinstance(date_val, datetime):
783 return date_val.date()
784
785 if isinstance(date_val, str):
786 return self._parse_date_string(date_val)
787
788 return date.today()
789
790 def _parse_date_string(self, date_str: str) -> date:
791 """Вспомогательный метод для парсинга строки даты."""
792 try:
793 return datetime.fromisoformat(date_str.replace('Z', '+00:00')).date()
794 except ValueError:
795 for fmt in ["%Y-%m-%d", "%d.%m.%Y"]:
796 try:
797 return datetime.strptime(date_str, fmt).date()
798 except ValueError:
799 continue
800 return date.today()
