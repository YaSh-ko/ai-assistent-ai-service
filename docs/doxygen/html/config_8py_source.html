272 def DATABASE_CONFIG(self) -> Dict[str, Any]:
273 """
274 @brief Возвращает словарь с параметрами подключения ко всем БД.
275
276 Если задан POSTGRES_URL или DATABASE_URL, парсит из него host/port/db/user/password
277 и перезаписывает соответствующие поля.
278
279 @return Словарь с ключами host, port, database, user, password,
280 neo4j_uri, chroma_host, milvus_host и т.д.
281 """
282 db_url = self.POSTGRES_URL or self.DATABASE_URL
283
284 config = {
285 "host": self.POSTGRES_HOST,
286 "port": self.POSTGRES_PORT,
287 "database": self.POSTGRES_DB,
288 "user": self.POSTGRES_USER,
289 "password": self.POSTGRES_PASSWORD,
290 "pool_size": 10,
291 "min_size": 10,
292 "max_size": 20,
293 "timeout": 30,
294 "neo4j_uri": self.NEO4J_URI,
295 "neo4j_user": self.NEO4J_USERNAME,
296 "neo4j_password": self.NEO4J_PASSWORD,
297 "chroma_host": self.CHROMA_SERVER_HOST,
298 "chroma_port": self.CHROMA_SERVER_PORT,
299 "chroma_ssl": self.CHROMA_SERVER_SSL,
300 "milvus_host": self.MILVUS_HOST,
301 "milvus_port": self.MILVUS_PORT,
302 "milvus_user": self.MILVUS_USER,
303 "milvus_password": self.MILVUS_PASSWORD,
304 "milvus_collection": self.MILVUS_COLLECTION,
305 }
306
307 # Parse DATABASE_URL if provided
308 if db_url:
309 try:
310 from urllib.parse import urlparse
311 parsed = urlparse(db_url)
312 if parsed.hostname: config["host"] = parsed.hostname
313 if parsed.port: config["port"] = parsed.port
314 if parsed.path.lstrip("/"): config["database"] = parsed.path.lstrip("/")
315 if parsed.username: config["user"] = parsed.username
316 if parsed.password: config["password"] = parsed.password
317 except Exception:
318 pass
319
320 return config
