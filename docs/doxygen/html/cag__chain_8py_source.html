92class CAGChain(BaseChain):
93 """
94 @brief Corrective Augmented Generation Chain.
95
96 Добавляет слой проверки качества поверх RAG:
97 1. Оценка релевантности найденных документов по score-порогу.
98 2. Генерация черновика ответа через LLMService.auto_select_and_generate().
99 3. Проверка качества: длина, маркеры неуверенности.
100 4. Коррекция при необходимости (до max_correction_attempts раз).
101 5. Финализация.

134 def build_graph(self) -> StateGraph:
135 """
136 @brief Строит и компилирует LangGraph-граф CAG-пайплайна.
137
138 Граф: retrieve → grade_relevance → generate_draft → check_quality
139 → (correct) correct_answer → check_quality (цикл до 2 раз)
140 → (finalize) finalize → END.
141
142 @return Скомпилированный граф.
143 """
144 workflow = StateGraph(CAGState)
145
146 # Определение узлов
147 workflow.add_node("retrieve", self.retrieve)
148 workflow.add_node("grade_relevance", self.grade_relevance)
149 workflow.add_node("generate_draft", self.generate_draft)
150 workflow.add_node("check_quality", self.check_quality)
151 workflow.add_node("correct_answer", self.correct_answer)
152 workflow.add_node("finalize", self.finalize)
153
154 # Определение связей
155 workflow.set_entry_point("retrieve")
156 workflow.add_edge("retrieve", "grade_relevance")
157 workflow.add_edge("grade_relevance", "generate_draft")
158 workflow.add_edge("generate_draft", "check_quality")
159
160 # Условный переход: коррекция или финализация
161 workflow.add_conditional_edges(
162 "check_quality",
163 self._should_correct,
164 {
165 "correct": "correct_answer",
166 "finalize": "finalize"
167 }
168 )
169 workflow.add_edge("correct_answer", "check_quality")
170 workflow.add_edge("finalize", END)
171
172 return workflow.compile()

174 def _should_correct(self, state: CAGState) -> str:
175 """
176 @brief Определяет, нужна ли коррекция ответа.
177
178 @param state Текущее состояние пайплайна.
179 @return "correct" если нужна коррекция и не исчерпан лимит попыток,
180 иначе "finalize".

182 if state.get("needs_correction", False):

184 return "correct"

191 async def retrieve(self, state: CAGState) -> CAGState:
192 """
193 @brief Шаг 1: поиск релевантных документов.
194
195 Генерирует эмбеддинг запроса и выполняет поиск через
196 search_provider или embedding_repository (fallback).
197
198 @param state Текущее состояние пайплайна.
199 @return Обновлённое состояние с search_results.
200 """
201 question = state["question"]
202 user_id = state.get("user_id")
203
204 logger.info(f"[CAG] Поиск для: '{question[:50]}...'")
205
206 search_results = []
207
208 try:
209 if self.embedding_service:
210 query_embedding = await self.embedding_service.generate_embedding(question)
211
212 if self.search_provider:
213 search_results = await self.search_provider.search(
214 query=question,
215 query_embedding=query_embedding,
216 top_k=10,
217 user_id=user_id
218 )
219 elif hasattr(self.dal, 'embedding_repo'):
220 search_results = await self.dal.embedding_repo.search_similar(
221 query_embedding=query_embedding,
222 top_k=10,
223 user_id=user_id
224 )
225 except Exception as e:
226 logger.error(f"[CAG] Ошибка поиска: {e}")
227
228 state["search_results"] = search_results
229 logger.info(f"[CAG] Найдено {len(search_results)} документов")
230
231 return state

237 async def grade_relevance(self, state: CAGState) -> CAGState:
238 """
239 @brief Шаг 2: оценка релевантности каждого документа.
240
241 Использует эвристику по score:
242 - score ≥ threshold → RELEVANT
243 - score ≥ threshold * 0.7 → PARTIALLY_RELEVANT
244 - иначе → NOT_RELEVANT
245
246 Оставляет только RELEVANT и PARTIALLY_RELEVANT документы (топ-5).
247
248 @param state Текущее состояние пайплайна.
249 @return Обновлённое состояние с relevance_grades и filtered_results.
250 """
251 search_results = state.get("search_results", [])
252
253 logger.info(f"[CAG] Оценка релевантности {len(search_results)} документов")
254
255 relevance_grades = []
256 filtered_results = []
257
258 for doc in search_results:
259
260 # Быстрая эвристика по score
261 score = doc.get("final_score", doc.get("score", 0))
262
263 if score >= self.relevance_threshold:
264 grade = RelevanceGrade.RELEVANT
265 elif score >= self.relevance_threshold * 0.7:
266 grade = RelevanceGrade.PARTIALLY_RELEVANT
267 else:
268 grade = RelevanceGrade.NOT_RELEVANT
269
270 relevance_grades.append({
271 "doc_id": doc.get("id", ""),
272 "grade": grade.value,
273 "score": score
274 })
275
276 # Оставляем только релевантные
277 if grade in [RelevanceGrade.RELEVANT, RelevanceGrade.PARTIALLY_RELEVANT]:
278 filtered_results.append(doc)
279
280 state["relevance_grades"] = relevance_grades
281 state["filtered_results"] = filtered_results[:5] # Топ-5
282
283 logger.info(f"[CAG] Отфильтровано: {len(filtered_results)} релевантных документов")
284
285 return state

293 @brief Шаг 3: генерация черновика ответа.

337 def _build_context(self, results: List[Dict[str, Any]]) -> str:
338 """Построение контекста из результатов поиска."""
339 if not results:
340 return "Релевантных записей не найдено."
341
342 parts = []
343 for i, result in enumerate(results, 1):
344 title = result.get('title', '')
345 content = result.get('description', result.get('page_content', ''))
346 date = result.get('event_date', '')
347
348 parts.append(f"""[Запись {i}]
349Дата: {date}
350Заголовок: {title}
351Содержание: {content}
352""")
353
354 return "\n".join(parts)

360 async def check_quality(self, state: CAGState) -> CAGState:
361 """
362 @brief Шаг 4: проверка качества черновика.
363
364 Проверяет:
365 1. Ответ не пустой и не слишком короткий (< 20 символов).
366 2. Ответ не содержит маркеры неуверенности при наличии контекста.
367
368 @param state Текущее состояние пайплайна.
369 @return Обновлённое состояние с needs_correction и correction_feedback.
370 """
371 draft_answer = state.get("draft_answer", "")
372
373 logger.info("[CAG] Проверка качества ответа")
374
375 correction_feedback = []
376
377 # Проверка 1: Ответ не пустой
378 if not draft_answer or len(draft_answer.strip()) < 20:
379 correction_feedback.append("Ответ слишком короткий")
380
381 # Проверка 2: Ответ содержит отказ или неуверенность
382 uncertainty_markers = [
383 "не могу помочь",
384 "не понимаю",
385 "ошибка",
386 "не знаю"
387 ]
388
389 has_uncertainty = any(
390 marker in draft_answer.lower()
391 for marker in uncertainty_markers
392 )
393
394 if has_uncertainty and state.get("filtered_results"):
395 # Есть контекст, но модель не уверена
396 correction_feedback.append("Есть контекст, но ответ содержит неуверенность")
397
398 # Решаем, нужна ли коррекция
399 needs_correction = len(correction_feedback) > 0
400 attempts = state.get("correction_attempts", 0)
401
402 state["needs_correction"] = needs_correction and attempts < self.max_correction_attempts
403 state["correction_feedback"] = correction_feedback
404 state["correction_attempts"] = attempts
405
406 if needs_correction:
407 logger.info(f"[CAG] Требуется коррекция: {correction_feedback}")
408
409 return state
