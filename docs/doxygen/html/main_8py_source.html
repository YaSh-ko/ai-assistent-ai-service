23def setup_paths():
24 """
25 @brief Добавляет корневую директорию проекта в sys.path.
26
27 Нужно, чтобы Python мог найти общие модули (common/), которые
28 лежат выше текущего пакета. Ищет родительскую папку, содержащую
29 директорию 'common', и добавляет её в начало пути.
30
31 @return Путь к корню проекта или None, если не найден.
32 """
33 current = Path(__file__).resolve()
34 for parent in current.parents:
35 if (parent / "common").exists():
36 root_str = str(parent)
37 if root_str not in sys.path:
38 sys.path.insert(0, root_str)
39 return parent
40 return None

168 @app.middleware("http")
169 async def prometheus_middleware(request: Request, call_next):
170 if request.url.path == "/metrics":
171 return await call_next(request)

173 endpoint = request.url.path
174 ai_active_requests.inc()
175 start = time.time()
176 status = "success"

178 response = await call_next(request)

186 duration = time.time() - start
187 ai_requests_total.labels(endpoint=endpoint, status=status).inc()
